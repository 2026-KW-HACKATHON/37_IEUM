title
type
description
target
period
status
capacity
requirements
resultType
evidence
deadline

{
  id: 1,
  title: "어르신 스마트폰 병원 예약 안내",
  type: "생활·디지털 안내",
  description: "어르신이 스마트폰으로 병원 예약을 할 수 있도록 안내 자료 제작",
  target: "어르신",
  period: "2026.10.10 ~ 2026.10.17",
  status: "모집 중",
  capacity: 1,
  requirements: "영상 또는 PDF 제작",
  resultType: "영상/PDF",
  evidence: "활동 결과물 및 활동일지",
  deadline: "2026.10.17"
